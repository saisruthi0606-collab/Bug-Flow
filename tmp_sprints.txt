import { useState, type FormEvent } from 'react'
import { useQuery, useQueryClient } from '@tanstack/react-query'
import Layout from '../components/Layout'
import { api } from '../lib/api'
import type { Issue, Project, Sprint } from '../lib/types'
export default function SprintsPage(){const qc=useQueryClient();const {data:sprints=[]}=useQuery<Sprint[]>({queryKey:['sprints'],queryFn:async()=> (await api.get('/api/sprints')).data});const {data:projects=[]}=useQuery<Project[]>({queryKey:['projects'],queryFn:async()=> (await api.get('/api/projects')).data});const {data:issues=[]}=useQuery<Issue[]>({queryKey:['issues'],queryFn:async()=> (await api.get('/api/issues',{params:{size:100}})).data});const blank={name:'',goal:'',start_date:'',end_date:'',status:'Planned',project_id:''};const [form,setForm]=useState<any>(blank);const [editing,setEditing]=useState<number|null>(null);const submit=async(e:FormEvent)=>{e.preventDefault();const payload={...form,project_id:Number(form.project_id)};editing?await api.put(`/api/sprints/${editing}`,payload):await api.post('/api/sprints',payload);setForm(blank);setEditing(null);qc.invalidateQueries({queryKey:['sprints']});qc.invalidateQueries({queryKey:['dashboard']})};return <Layout title="Sprints"><div className="grid gap-6 lg:grid-cols-[1fr_.65fr]"><section className="space-y-3">{sprints.map(s=><div key={s.id} className="rounded-xl border border-border bg-card p-5"><div className="flex justify-between"><h2 className="font-semibold">{s.name}</h2><select value={s.status} onChange={async e=>{await api.put(`/api/sprints/${s.id}`,{status:e.target.value});qc.invalidateQueries({queryKey:['sprints']})}}><option>Planned</option><option>Active</option><option>Completed</option></select></div><p className="mt-2 text-sm text-muted-foreground">{s.goal}</p><p className="mt-3 text-sm">{s.issue_count} issues • {s.start_date} to {s.end_date}</p><button onClick={()=>{setEditing(s.id);setForm({...s,project_id:String(s.project_id)})}} className="mt-3 text-sm text-primary">Edit</button><button onClick={async()=>{if(window.confirm('Delete sprint?')){await api.delete(`/api/sprints/${s.id}`);qc.invalidateQueries({queryKey:['sprints']})}}} className="ml-3 text-sm text-rose-500">Delete</button><div className="mt-3 flex flex-wrap gap-2">{issues.filter(i=>i.project_id===s.project_id).map(i=><button key={i.id} onClick={async()=>{if(i.sprint_id===s.id)await api.delete(`/api/sprints/${s.id}/issues/${i.id}`);else await api.post(`/api/sprints/${s.id}/issues/${i.id}`);qc.invalidateQueries({queryKey:['issues']});qc.invalidateQueries({queryKey:['sprints']})}} className="rounded bg-background px-2 py-1 text-xs">{i.sprint_id===s.id?'Remove':'Assign'} {i.title}</button>)}</div></div>)}</section><form onSubmit={submit} className="space-y-3 rounded-xl border border-border bg-card p-6"><h2 className="font-semibold">{editing?'Edit sprint':'Create sprint'}</h2><input required className="w-full rounded-xl border border-border bg-background p-2" placeholder="Name" value={form.name} onChange={e=>setForm({...form,name:e.target.value})}/><textarea className="w-full rounded-xl border border-border bg-background p-2" placeholder="Goal" value={form.goal||''} onChange={e=>setForm({...form,goal:e.target.value})}/><input required type="date" className="w-full rounded-xl border border-border bg-background p-2" value={form.start_date} onChange={e=>setForm({...form,start_date:e.target.value})}/><input required type="date" className="w-full rounded-xl border border-border bg-background p-2" value={form.end_date} onChange={e=>setForm({...form,end_date:e.target.value})}/><select required className="w-full rounded-xl border border-border bg-background p-2" value={form.project_id} onChange={e=>setForm({...form,project_id:e.target.value})}><option value="">Select project</option>{projects.map(p=><option key={p.id} value={p.id}>{p.project_name}</option>)}</select><button className="rounded-xl bg-primary px-4 py-2">{editing?'Save sprint':'Create sprint'}</button></form></div></Layout>}
